! ==============================================================================
! DEVICE:       SW-D3 (Remote Clinic - Distribution Switch)
! MODEL:        Cisco Catalyst 3560-24PS (Multilayer Layer-3 Switch)
! LAYER:        Distribution Layer (Remote Clinic Campus)
! UPLINK:       GigabitEthernet0/1 -> R2 GigabitEthernet0/0 (Routed Link /30)
! DOWNLINK:     FastEthernet0/1 -> SW-A5 (802.1Q Trunk)
! ROUTING:      OSPF Process 1 (Area 1 - Remote Clinic Stub/Non-backbone Area)
! ==============================================================================

enable
configure terminal

hostname SW-D3

! --- Disable DNS Lookup & Set Banner ---
no ip domain-lookup
banner motd ^C
====================================================================
  SW-D3: Remote Clinic Distribution Switch - VIT SCOPE CN Project
  Multilayer Layer-3 Aggregator for Remote Clinic Subnets (OSPF Area 1)
====================================================================
^C

! --- Management Credentials & Line Security ---
enable secret class
line con 0
 password cisco
 login
 logging synchronous
 exec-timeout 15 0
line vty 0 4
 password cisco
 login
 transport input ssh telnet
 exec-timeout 15 0
exit

! --- Enable Hardware Layer-3 IP Routing ---
ip routing

! --- VLAN Database Definition ---
vlan 10
 name Remote_Clinical
vlan 30
 name Remote_Admin
vlan 50
 name Remote_Server_Cache
vlan 60
 name Remote_VoIP
vlan 99
 name Management
vlan 999
 name Parking_Native
exit

! --- Spanning Tree Mode ---
spanning-tree mode pvst
spanning-tree vlan 10,30,50,60,99 root primary

! --- Switched Virtual Interfaces (SVIs) ---
! VLAN 10 (Remote Clinical): Subnet 172.16.4.64/27 (Mask 255.255.255.224)
interface Vlan10
 description SVI_Gateway_for_Remote_Clinical
 ip address 172.16.4.65 255.255.255.224
 no shutdown
exit

! VLAN 30 (Remote Admin): Subnet 172.16.4.96/27 (Mask 255.255.255.224)
interface Vlan30
 description SVI_Gateway_for_Remote_Admin
 ip address 172.16.4.97 255.255.255.224
 no shutdown
exit

! VLAN 60 (Remote VoIP): Subnet 172.16.4.128/27 (Mask 255.255.255.224)
interface Vlan60
 description SVI_Gateway_for_Remote_VoIP
 ip address 172.16.4.129 255.255.255.224
 no shutdown
exit

! VLAN 50 (Remote Server Cache): Subnet 172.16.4.160/28 (Mask 255.255.255.240)
interface Vlan50
 description SVI_Gateway_for_Remote_Server_Cache
 ip address 172.16.4.161 255.255.255.240
 no shutdown
exit

! VLAN 99 (Management): Subnet 172.16.4.176/28 (Mask 255.255.255.240)
interface Vlan99
 description SVI_Gateway_for_Remote_Management
 ip address 172.16.4.177 255.255.255.240
 no shutdown
exit

! --- Downlink Trunk to SW-A5 ---
interface FastEthernet0/1
 description Trunk_Downlink_to_SW-A5_Fa0/24
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,30,50,60,99
 no shutdown
exit

! --- Routed Uplink to Core Router R2 (Point-to-Point /30 Subnet) ---
interface GigabitEthernet0/1
 description Routed_Uplink_to_R2_Gig0/0
 no switchport
 ip address 172.16.5.9 255.255.255.252
 no shutdown
exit

! --- Dynamic Routing: OSPF Process 1 (Area 1 - Remote Clinic) ---
router ospf 1
 router-id 172.16.5.9
 passive-interface default
 no passive-interface GigabitEthernet0/1
 ! Advertise Remote Clinic subnets into Area 1
 network 172.16.4.64 0.0.0.31 area 1
 network 172.16.4.96 0.0.0.31 area 1
 network 172.16.4.128 0.0.0.31 area 1
 network 172.16.4.160 0.0.0.15 area 1
 network 172.16.4.176 0.0.0.15 area 1
 ! Transit link to R2
 network 172.16.5.8 0.0.0.3 area 1
exit

! Default route upstream to Core Router R2
ip route 0.0.0.0 0.0.0.0 172.16.5.10

end
write memory

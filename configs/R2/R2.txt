! ==============================================================================
! DEVICE:       R2 (Remote Clinic - Core Router / ABR)
! MODEL:        Cisco 2911 Integrated Services Router
! MODULES:      HWIC-2T installed in Slot 0 (provides Serial0/0/0 & Serial0/0/1)
! WAN PRIMARY:  Serial0/0/0 (DTE) -> R1 Se0/0/0 [OSPF Area 0 Backbone, AD 110]
! WAN BACKUP:   Serial0/0/1 (DTE) -> R1 Se0/0/1 [EIGRP AS 100, AD adjusted to 120]
! LAN LINK:     GigabitEthernet0/0 -> SW-D3 Gi0/1 [OSPF Area 1 - Remote Clinic]
! ROLE:         OSPF Area Border Router (ABR) connecting Area 0 and Area 1
! ==============================================================================

enable
configure terminal

hostname R2

! --- Disable DNS Lookup & Set Banner ---
no ip domain-lookup
banner motd ^C
====================================================================
  R2: Remote Clinic Core Router (ABR) - VIT SCOPE CN Project
  OSPF Area Border Router (Area 0 <-> Area 1) with EIGRP Backup
====================================================================
^C

! --- Management Credentials & Line Security ---
enable secret class
line con 0
 password cisco
 login
 logging synchronous
 exec-timeout 15 0
line vty 0 4
 password cisco
 login
 transport input ssh telnet
 exec-timeout 15 0
exit

! --- Loopback0 Interface (OSPF Router ID) ---
interface Loopback0
 description OSPF_Router_ID_and_Management
 ip address 172.16.4.201 255.255.255.255
 no shutdown
exit

! --- LAN Point-to-Point Uplink to Remote Distribution Switch (SW-D3) ---
interface GigabitEthernet0/0
 description Link_to_SW-D3_Gi0/1_Area1
 ip address 172.16.5.10 255.255.255.252
 no shutdown
exit

! --- Primary WAN Link (Serial0/0/0 - DTE): OSPF Area 0 Backbone ---
interface Serial0/0/0
 description Primary_WAN_Link_to_R1_Se0/0/0_OSPF_Area0
 ip address 172.16.4.194 255.255.255.252
 bandwidth 2048
 no shutdown
exit

! --- Secondary WAN Link (Serial0/0/1 - DTE): Standby EIGRP AS 100 ---
interface Serial0/0/1
 description Standby_WAN_Link_to_R1_Se0/0/1_EIGRP_Backup
 ip address 172.16.4.198 255.255.255.252
 bandwidth 1544
 no shutdown
exit

! --- Dynamic Routing: Multi-Area OSPF Process 1 (ABR Role) ---
router ospf 1
 router-id 172.16.4.201
 ! Area 0: WAN Primary Serial Link to R1
 network 172.16.4.192 0.0.0.3 area 0
 ! Area 1: Remote Clinic Campus Network
 network 172.16.4.201 0.0.0.0 area 1       ! Loopback0 in Area 1
 network 172.16.5.8 0.0.0.3 area 1         ! Link to SW-D3 in Area 1
exit

! --- Dynamic Routing: EIGRP AS 100 (Secondary Standby Path) ---
router eigrp 100
 no auto-summary
 network 172.16.4.196 0.0.0.3              ! Secondary Serial Se0/0/1
 network 172.16.5.8 0.0.0.3                ! Link to SW-D3
 ! Remote Clinic subnets for secondary advertisement
 network 172.16.4.64 0.0.0.31
 network 172.16.4.96 0.0.0.31
 network 172.16.4.128 0.0.0.31
 network 172.16.4.160 0.0.0.15
 network 172.16.4.176 0.0.0.15
 ! Raise AD: internal=120, external=170
 distance eigrp 120 170
exit

! --- Quality of Service (QoS): Modular QoS CLI (MQC) ---
class-map match-any VITALS-VOICE
 match ip dscp ef
exit

class-map match-any IMAGING
 match ip dscp af31
exit

policy-map HOSPITAL-QOS
 class VITALS-VOICE
  priority percent 30
 class IMAGING
  bandwidth percent 40
 class class-default
  fair-queue
exit

! Apply QoS policy egress on WAN links
interface Serial0/0/0
 service-policy output HOSPITAL-QOS
exit

interface Serial0/0/1
 service-policy output HOSPITAL-QOS
exit

end
write memory
